solution
// Fetch the reflection cube.
walk_to 8.5 5.3
grab
// Over to the catcher's line, and over the beam with the cube.
walk_to 3.5 3.6
jump
// The cube goes down south of the beam, facing north: the beam goes up
// to the catcher.
walk_to 3.5 1.5
look 3.5 1.3 2.9
wait 0.4
use
wait 0.5
// Over the beam, to the door.
walk_to 5.5 1.6
jump
walk_to 12.5 10.8
